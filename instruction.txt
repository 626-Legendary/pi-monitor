# 快速操作；完整说明见 README.md

# 构建并启动
docker compose up -d --build

# 可选：安装 / 更新宿主机 SSH 只读采集服务
sudo sh host/install.sh

# 查看日志
docker compose logs -f --tail=100

# 停止 / 移除容器（保留镜像和命名卷）
docker compose stop
docker compose down
